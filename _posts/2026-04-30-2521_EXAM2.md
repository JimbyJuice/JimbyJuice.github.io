## `bst.h`

```c
#ifndef BST_H
#define BST_H

// The node structure for the BST
struct node {
    int value;
    struct node *left;
    struct node *right;
};

// Creates and returns a new node with the given value
struct node *newNode(int value);

// Inserts a value into the BST and returns the root
struct node *bstInsert(struct node *t, int value);

// Searches for a value in the BST
// Returns 1 if found, 0 otherwise
int bstSearch(struct node *t, int value);

// Frees all memory associated with the BST
void bstFree(struct node *t);

// Prints the BST in-order
void bstPrint(struct node *t);

#endif
```

## `bst.c`

```c
#include <stdio.h>
#include <stdlib.h>
#include "bst.h"

struct node *newNode(int value) {
    struct node *n = malloc(sizeof(struct node));
    if (n == NULL) {
        fprintf(stderr, "error: out of memory\n");
        exit(1);
    }
    n->value = value;
    n->left  = NULL;
    n->right = NULL;
    return n;
}

struct node *bstInsert(struct node *t, int value) {
    if (t == NULL) return newNode(value);
    if (value < t->value)
        t->left  = bstInsert(t->left,  value);
    else if (value > t->value)
        t->right = bstInsert(t->right, value);
    // duplicate values are ignored
    return t;
}

int bstSearch(struct node *t, int value) {
    if (t == NULL)          return 0;
    if (value == t->value)  return 1;
    if (value < t->value)   return bstSearch(t->left,  value);
    return                         bstSearch(t->right, value);
}

void bstFree(struct node *t) {
    if (t == NULL) return;
    bstFree(t->left);
    bstFree(t->right);
    free(t);
}

void bstPrint(struct node *t) {
    if (t == NULL) return;
    bstPrint(t->left);
    printf("%d ", t->value);
    bstPrint(t->right);
}
```

**Theory hints:**

- Q5b: Stable — the relative order of equal keys (6,A), (6,C), (6,F) is preserved from input.

**Practical hints:**

- Q7: The O(log n) solution adapts binary search — if `arr[mid] > arr[right]`, the rotation point is in the right half.
- Q8: Insertion sort is the natural fit — each element moves at most k positions, giving O(nk). But the O(n log k) solution uses a min-heap of size k.
- Q9: Three pointer approach — lo, mid, hi. This is the Lomuto partition generalised to three-way.
- Q10: Naive = repeatedly scan all k arrays for the minimum. O(n log k) = use a min-heap tracking one element from each array.
